# 50 Cent: Bulletproof – PC Port

cmake_minimum_required(VERSION 3.25)
project(Bulletproof VERSION 0.8.0 DESCRIPTION "Native Vulkan PC Port for Windows and Linux" LANGUAGES C CXX)

set(BULLETPROOF_PUBLIC_NAME "50 Cent: Bulletproof – PC Port")
configure_file(src/common/bulletproof_metadata.h.in bulletproof_metadata.h @ONLY)
set_property(SOURCE src/common/game_info.cpp APPEND PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(BULLETPROOF_PORTABLE_ONLY "Build the portable host and recompiler without an XBE" OFF)
set(BULLETPROOF_PORTABLE_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/game.toml" CACHE FILEPATH
    "Verified Bulletproof title metadata for the portable host")
if(BULLETPROOF_PORTABLE_ONLY)
    if(WIN32)
        set(XBE_WINDOWS_PORTABLE ON)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(XBE_LINUX_PORTABLE_EXPERIMENT ON)
    else()
        message(FATAL_ERROR "The portable host supports Linux and Windows.")
    endif()
endif()

# Use the bundled, version-matched Vulkan runtime on both hosts.
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
if(NOT EXISTS "${SDK}/linux/cmake/XbeRecomp.cmake")
    message(FATAL_ERROR "The bundled Vulkan SDK is missing. Restore sdk/linux from this checkout.")
endif()
set(XBE_XDK_DIR "${SDK}/xdk")
include("${SDK}/linux/cmake/XbeRecomp.cmake")
file(GLOB XBE_HOST_SOURCES CONFIGURE_DEPENDS src/common/*.cpp)

if(NOT BULLETPROOF_PORTABLE_ONLY)
    # The static target uses the partition count and disc revision from the local inputs.
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS config/game.toml)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config/game.toml")
        file(STRINGS config/game.toml GAME_PARTITIONS REGEX "^partitions *=")
        string(REGEX MATCH "[0-9]+" GAME_PARTITIONS "${GAME_PARTITIONS}")
    endif()
    if(NOT GAME_PARTITIONS)
        set(GAME_PARTITIONS 8)
    endif()
    add_xbe_image(Bulletproof default.xbe game.toml ${GAME_PARTITIONS} game)
else()
    set_target_properties(xbe_symbols PROPERTIES EXCLUDE_FROM_ALL ON)
endif()

if((CMAKE_SYSTEM_NAME STREQUAL "Linux" AND XBE_LINUX_PORTABLE_EXPERIMENT) OR XBE_WINDOWS_PORTABLE)
    # SHA-1 of the supported default.xbe, verified against the local disc revision.
    set(portable_exclude)
    if(NOT BULLETPROOF_PORTABLE_ONLY)
        set(portable_exclude EXCLUDE_FROM_ALL)
    endif()
    add_xbe_portable(Bulletproof_portable "${BULLETPROOF_PORTABLE_CONFIG}"
        "0abb01d5e8b323f13d26ac98b619e42220795093" game ${portable_exclude})
endif()

if(WIN32)
    enable_language(RC)
    configure_file(src/common/bulletproof_version.rc.in bulletproof_version.rc @ONLY)
    foreach(target IN ITEMS Bulletproof Bulletproof_portable)
        if(TARGET ${target})
            target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/bulletproof_version.rc")
        endif()
    endforeach()
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    option(BUILD_TESTING "Build Bulletproof regression tests" OFF)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(native_menu_state_test tests/native_menu_state.cpp)
        target_include_directories(native_menu_state_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME native_menu_state COMMAND native_menu_state_test)

        add_executable(stair_contact_test tests/stair_contact_test.cpp)
        target_include_directories(stair_contact_test PRIVATE src)
        add_test(NAME stair_contact COMMAND stair_contact_test)
        add_executable(framerate_timing_test tests/framerate_timing_test.cpp
            src/game/framerate_timing.cpp)
        target_include_directories(framerate_timing_test PRIVATE "${SDK}/linux/runtime")
        add_test(NAME framerate_timing COMMAND framerate_timing_test)
        add_executable(desktop_input_test tests/desktop_input_test.cpp src/game/prompt_bindings.cpp
            "${SDK}/linux/runtime/core/desktop_input.cpp")
        target_include_directories(desktop_input_test PRIVATE src "${SDK}/linux/runtime")
        target_link_libraries(desktop_input_test PRIVATE SDL3::SDL3-static)
        add_test(NAME desktop_input COMMAND desktop_input_test)
        add_executable(l3_crouch_test tests/l3_crouch_test.cpp src/game/input_system.cpp
            "${SDK}/linux/runtime/core/desktop_input.cpp")
        target_include_directories(l3_crouch_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SDK}/linux/runtime")
        target_link_libraries(l3_crouch_test PRIVATE tomlplusplus::tomlplusplus SDL3::SDL3-static)
        add_test(NAME l3_crouch COMMAND l3_crouch_test)
        add_executable(video_frame_snapshot_test tests/video_frame_snapshot.cpp)
        target_include_directories(video_frame_snapshot_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME video_frame_snapshot COMMAND video_frame_snapshot_test)
        find_package(Python3 QUIET COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND)
            execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy"
                RESULT_VARIABLE VIDEO_ORDER_NUMPY OUTPUT_QUIET ERROR_QUIET)
            if(VIDEO_ORDER_NUMPY EQUAL 0)
                add_test(NAME video_frame_order COMMAND "${Python3_EXECUTABLE}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/tests/video_frame_order.py")
            endif()
        endif()
        add_executable(linux_disc_smoke tests/disc_smoke.cpp
            "${SDK}/linux/runtime/kernel/disc.cpp")
        target_include_directories(linux_disc_smoke PRIVATE "${SDK}/linux/runtime")
        add_executable(apu_mmio_test tests/apu_mmio.cpp
            "${SDK}/linux/runtime/audio/apu_clock.cpp")
        target_include_directories(apu_mmio_test PRIVATE "${SDK}/linux/runtime")
        target_link_libraries(apu_mmio_test PRIVATE pthread)
        if(TARGET recompile_Bulletproof)
            add_executable(dsound_voice_test tests/dsound_voice.cpp src/game/linux_dsound.cpp)
            add_dependencies(dsound_voice_test recompile_Bulletproof)
            target_include_directories(dsound_voice_test PRIVATE "${SDK}/linux/runtime"
                "${CMAKE_CURRENT_BINARY_DIR}/generated/game")
            target_compile_options(dsound_voice_test PRIVATE -ffunction-sections -fdata-sections)
            target_link_options(dsound_voice_test PRIVATE -fuse-ld=lld -Wl,--gc-sections)
            target_link_libraries(dsound_voice_test PRIVATE pthread)
            add_executable(vertex_program_test tests/vertex_program.cpp src/game/linux_graphics.cpp
                "${SDK}/linux/runtime/gpu/shaders.cpp")
            add_dependencies(vertex_program_test recompile_Bulletproof)
            target_include_directories(vertex_program_test PRIVATE "${SDK}/linux/runtime"
                "${CMAKE_CURRENT_BINARY_DIR}/generated/game")
            target_compile_options(vertex_program_test PRIVATE -ffunction-sections -fdata-sections)
            target_link_options(vertex_program_test PRIVATE -fuse-ld=lld -Wl,--gc-sections)
            target_link_libraries(vertex_program_test PRIVATE pthread)
            add_test(NAME vertex_program COMMAND vertex_program_test)
            add_test(NAME dsound_voice COMMAND dsound_voice_test)
        endif()
        add_executable(worker_sync_test tests/worker_sync.cpp
            "${SDK}/linux/runtime/kernel/threads.cpp")
        target_include_directories(worker_sync_test PRIVATE "${SDK}/linux/runtime")
        target_link_libraries(worker_sync_test PRIVATE pthread)
        add_test(NAME worker_sync COMMAND worker_sync_test)
        add_executable(vblank_wait_test tests/vblank_wait.cpp)
        target_include_directories(vblank_wait_test PRIVATE "${SDK}/linux/runtime")
        target_link_libraries(vblank_wait_test PRIVATE pthread)
        add_test(NAME vblank_wait COMMAND vblank_wait_test)
        add_executable(spirv_bindings_test tests/spirv_bindings.cpp
            "${SDK}/linux/runtime/gpu/spirv.cpp" "${SDK}/linux/runtime/gpu/shaders.cpp")
        target_include_directories(spirv_bindings_test PRIVATE "${SDK}/linux/runtime" ${XBE_SHADER_INCLUDES})
        target_compile_definitions(spirv_bindings_test PRIVATE ${XBE_SHADER_DEFINES})
        target_compile_options(spirv_bindings_test PRIVATE -ffunction-sections -fdata-sections)
        target_link_options(spirv_bindings_test PRIVATE -fuse-ld=lld -Wl,--gc-sections)
        target_link_libraries(spirv_bindings_test PRIVATE ${XBE_SHADER_LIBS} pthread dl)
        add_test(NAME spirv_bindings COMMAND spirv_bindings_test "${CMAKE_CURRENT_BINARY_DIR}/tests/spirv-cache")
        add_test(NAME apu_mmio COMMAND apu_mmio_test)
        add_test(NAME linux_disc_smoke COMMAND linux_disc_smoke
            "${CMAKE_CURRENT_BINARY_DIR}/tests/disc")
        add_executable(savegame_portable_test tests/savegame_portable_test.cpp
            "${SDK}/linux/runtime/kernel/files.cpp" "${SDK}/linux/runtime/app/userdata.cpp")
        target_include_directories(savegame_portable_test PRIVATE "${SDK}/linux/runtime")
        target_link_libraries(savegame_portable_test PRIVATE pthread)
        add_test(NAME savegame_portable COMMAND savegame_portable_test)
    endif()
endif()
